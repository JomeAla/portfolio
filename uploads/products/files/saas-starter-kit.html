4. Payment Integration (Paystack)

Setup
// config/services.php 'paystack' => [ 'secret' => env('PAYSTACK_SECRET_KEY'), 'public' => env('PAYSTACK_PUBLIC_KEY'), ], // .env PAYSTACK_SECRET_KEY=sk_live_xxxx PAYSTACK_PUBLIC_KEY=pk_live_xxxx
Payment Controller
// app/Http/Controllers/PaymentController.php public function initialize(Request $request) { $user = auth()->user(); $amount = $request->amount * 100; // kobo $response = Http::withToken(config('services.paystack.secret')) ->post('https://api.paystack.co/transaction/initialize', [ 'email' => $user->email, 'amount' => $amount, 'callback_url' => route('payment.callback') ]); return $response->json(); }

5. Subscription Billing

Plans Configuration
// database/migrations/xxxx_create_plans_table.php Schema::create('plans', function (Blueprint $table) { $table->id(); $table->string('name'); $table->string('slug')->unique(); $table->decimal('price', 10, 2); $table->integer('interval_days'); $table->text('features')->nullable(); $table->boolean('is_active')->default(true); $table->timestamps(); }); // Seed data: Basic ₦5000/mo, Pro ₦15000/mo
Subscription Model
// app/Models/Subscription.php class Subscription extends Model { public function user() { return $this->belongsTo(User::class); } public function plan() { return $this->belongsTo(Plan::class); } public function isActive() { return $this->status === 'active' && $this->expires_at > now(); } }
